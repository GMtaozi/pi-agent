## 1. 数据库选型

| 用途 | 技术 | 说明 |
|------|------|------|
| 主数据库 | SQLite 3 | 单机零运维，JSON 支持好 |
| 缓存 | Redis | 热点数据、会话缓存 |
| 向量检索 | SQLite + json1 扩展 | Phase 1 使用，后续可升级 Qdrant |

## 3. 核心表设计

### 3.1 用户表（users）

```sql
CREATE TABLE users (
  id TEXT PRIMARY KEY,
  email TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  name TEXT NOT NULL,
  avatar TEXT,
  role TEXT DEFAULT 'user' CHECK (role IN ('user', 'admin')),
  plan TEXT DEFAULT 'free' CHECK (plan IN ('free', 'pro', 'enterprise')),
  status TEXT DEFAULT 'active' CHECK (status IN ('active', 'inactive', 'suspended')),
  api_key_encrypted TEXT,           -- 加密存储的 API Key
  preferences TEXT,                 -- JSON 格式用户偏好
  last_login_at TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_plan ON users(plan);
```

### 3.2 智能体表（agents）

```sql
CREATE TABLE agents (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  name TEXT NOT NULL,
  description TEXT,
  model TEXT NOT NULL DEFAULT 'gpt-4o',
  system_prompt TEXT,
  temperature REAL DEFAULT 0.7,
  max_tokens INTEGER DEFAULT 2000,
  knowledge_base_ids TEXT,          -- JSON 数组
  tool_names TEXT,                  -- JSON 数组
  tone TEXT DEFAULT 'professional',
  language TEXT DEFAULT 'zh',
  status TEXT DEFAULT 'active' CHECK (status IN ('active', 'inactive', 'draft')),
  avatar TEXT,
  metadata TEXT,                    -- JSON 扩展字段
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_agents_user_id ON agents(user_id);
CREATE INDEX idx_agents_status ON agents(status);
CREATE INDEX idx_agents_model ON agents(model);
```

### 3.3 会话表（sessions）

```sql
CREATE TABLE sessions (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  agent_id TEXT NOT NULL,
  title TEXT DEFAULT '新对话',
  status TEXT DEFAULT 'active' CHECK (status IN ('active', 'archived', 'deleted')),
  mode TEXT DEFAULT 'standard' CHECK (mode IN ('standard', 'debug')),
  total_tokens INTEGER DEFAULT 0,
  total_cost REAL DEFAULT 0,
  message_count INTEGER DEFAULT 0,
  metadata TEXT,                    -- JSON 扩展字段
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (agent_id) REFERENCES agents(id)
);

CREATE INDEX idx_sessions_user_id ON sessions(user_id);
CREATE INDEX idx_sessions_agent_id ON sessions(agent_id);
CREATE INDEX idx_sessions_status ON sessions(status);
CREATE INDEX idx_sessions_updated_at ON sessions(updated_at);
```

### 3.4 消息表（messages）

```sql
CREATE TABLE messages (
  id TEXT PRIMARY KEY,
  session_id TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('user', 'assistant', 'system', 'tool')),
  content TEXT NOT NULL,
  tokens INTEGER DEFAULT 0,
  cost REAL DEFAULT 0,
  tool_name TEXT,                   -- 工具调用时的工具名
  tool_input TEXT,                  -- 工具调用输入
  tool_output TEXT,                 -- 工具调用输出
  metadata TEXT,                    -- JSON 扩展字段
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (session_id) REFERENCES sessions(id)
);

CREATE INDEX idx_messages_session_id ON messages(session_id);
CREATE INDEX idx_messages_created_at ON messages(created_at);
CREATE INDEX idx_messages_session_created ON messages(session_id, created_at);
```

## 4. 监控相关表

### 4.1 执行记录表（executions）

```sql
CREATE TABLE executions (
  id TEXT PRIMARY KEY,
  agent_id TEXT NOT NULL,
  session_id TEXT,
  user_id TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed', 'stopped')),
  started_at TEXT NOT NULL,
  completed_at TEXT,
  duration_ms INTEGER,
  total_tokens INTEGER DEFAULT 0,
  prompt_tokens INTEGER DEFAULT 0,
  completion_tokens INTEGER DEFAULT 0,
  total_cost REAL DEFAULT 0,
  model TEXT NOT NULL,
  error_message TEXT,
  metadata TEXT,                    -- JSON 扩展字段
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (agent_id) REFERENCES agents(id),
  FOREIGN KEY (session_id) REFERENCES sessions(id),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_executions_agent_id ON executions(agent_id);
CREATE INDEX idx_executions_session_id ON executions(session_id);
CREATE INDEX idx_executions_user_id ON executions(user_id);
CREATE INDEX idx_executions_status ON executions(status);
CREATE INDEX idx_executions_started_at ON executions(started_at);
CREATE INDEX idx_executions_agent_started ON executions(agent_id, started_at);
CREATE INDEX idx_executions_user_started ON executions(user_id, started_at);
```

### 4.2 节点执行记录表（node_executions）

```sql
CREATE TABLE node_executions (
  id TEXT PRIMARY KEY,
  execution_id TEXT NOT NULL,
  node_id TEXT NOT NULL,
  node_type TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'running' CHECK (status IN ('running', 'completed', 'failed', 'skipped')),
  started_at TEXT NOT NULL,
  completed_at TEXT,
  duration_ms INTEGER,
  tokens INTEGER DEFAULT 0,
  prompt_tokens INTEGER DEFAULT 0,
  completion_tokens INTEGER DEFAULT 0,
  cost REAL DEFAULT 0,
  input TEXT,                       -- JSON 格式
  output TEXT,                      -- JSON 格式
  error TEXT,
  parent_node_id TEXT,              -- 父节点 ID（用于嵌套执行）
  FOREIGN KEY (execution_id) REFERENCES executions(id)
);

CREATE INDEX idx_node_executions_execution_id ON node_executions(execution_id);
CREATE INDEX idx_node_executions_node_id ON node_executions(node_id);
CREATE INDEX idx_node_executions_status ON node_executions(status);
CREATE INDEX idx_node_executions_node_type ON node_executions(node_type);
```

### 4.3 Token 消耗记录表（token_usage）

```sql
CREATE TABLE token_usage (
  id TEXT PRIMARY KEY,
  execution_id TEXT,
  session_id TEXT,
  agent_id TEXT NOT NULL,
  user_id TEXT NOT NULL,
  model TEXT NOT NULL,
  prompt_tokens INTEGER NOT NULL,
  completion_tokens INTEGER NOT NULL,
  total_tokens INTEGER NOT NULL,
  cost REAL NOT NULL,
  timestamp TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (execution_id) REFERENCES executions(id),
  FOREIGN KEY (session_id) REFERENCES sessions(id),
  FOREIGN KEY (agent_id) REFERENCES agents(id),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_token_usage_execution_id ON token_usage(execution_id);
CREATE INDEX idx_token_usage_agent_id ON token_usage(agent_id);
CREATE INDEX idx_token_usage_user_id ON token_usage(user_id);
CREATE INDEX idx_token_usage_timestamp ON token_usage(timestamp);
CREATE INDEX idx_token_usage_model ON token_usage(model);
CREATE INDEX idx_token_usage_user_timestamp ON token_usage(user_id, timestamp);
```

### 4.4 每日统计表（daily_stats）

```sql
CREATE TABLE daily_stats (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  agent_id TEXT,
  date TEXT NOT NULL,                -- 日期 YYYY-MM-DD
  total_executions INTEGER DEFAULT 0,
  successful_executions INTEGER DEFAULT 0,
  failed_executions INTEGER DEFAULT 0,
  total_tokens INTEGER DEFAULT 0,
  total_cost REAL DEFAULT 0,
  avg_duration_ms REAL DEFAULT 0,
  UNIQUE(user_id, agent_id, date),
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (agent_id) REFERENCES agents(id)
);

CREATE INDEX idx_daily_stats_user_date ON daily_stats(user_id, date);
CREATE INDEX idx_daily_stats_agent_date ON daily_stats(agent_id, date);
CREATE INDEX idx_daily_stats_date ON daily_stats(date);
```

## 5. 知识库相关表

### 5.1 知识库表（knowledge_bases）

```sql
CREATE TABLE knowledge_bases (
  id TEXT PRIMARY KEY,
  user_id TEXT NOT NULL,
  name TEXT NOT NULL,
  description TEXT,
  embedding_model TEXT DEFAULT 'text-embedding-3-small',
  chunk_size INTEGER DEFAULT 500,
  chunk_overlap INTEGER DEFAULT 50,
  status TEXT DEFAULT 'active' CHECK (status IN ('active', 'inactive', 'indexing')),
  document_count INTEGER DEFAULT 0,
  chunk_count INTEGER DEFAULT 0,
  metadata TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_knowledge_bases_user_id ON knowledge_bases(user_id);
CREATE INDEX idx_knowledge_bases_status ON knowledge_bases(status);
```

### 5.2 文档表（documents）

```sql
CREATE TABLE documents (
  id TEXT PRIMARY KEY,
  knowledge_base_id TEXT NOT NULL,
  user_id TEXT NOT NULL,
  name TEXT NOT NULL,
  type TEXT NOT NULL,                -- pdf, docx, md, txt, html, excel
  size INTEGER NOT NULL,
  path TEXT NOT NULL,                -- 文件存储路径
  status TEXT DEFAULT 'pending' CHECK (status IN ('pending', 'indexing', 'indexed', 'failed')),
  chunk_count INTEGER DEFAULT 0,
  metadata TEXT,                    -- JSON 扩展字段
  error_message TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (knowledge_base_id) REFERENCES knowledge_bases(id),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_documents_kb_id ON documents(knowledge_base_id);
CREATE INDEX idx_documents_status ON documents(status);
CREATE INDEX idx_documents_user_id ON documents(user_id);
```

### 5.3 文档分块表（document_chunks）

CREATE TABLE document_chunks (

  embedding TEXT,                   -- JSON 数组（向量）
  metadata TEXT,                    -- JSON 扩展字段
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  FOREIGN KEY (document_id) REFERENCES documents(id),
  FOREIGN KEY (knowledge_base_id) REFERENCES knowledge_bases(id)

CREATE INDEX idx_document_chunks_document_id ON document_chunks(document_id);
CREATE INDEX idx_document_chunks_kb_id ON document_chunks(knowledge_base_id);
